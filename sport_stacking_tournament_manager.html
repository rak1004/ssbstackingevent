구글 시트 연동 가이드 (예선/본선 시트 각각 CSV 웹 게시)

  1. 스프레드시트에 예선 시트와 본선 시트를 만듭니다.
  2. 각 시트를 [파일] > [공유] > [웹에 게시]하여 CSV 링크를 복사하세요.
  3. 예선 컬럼 순서: 학년, 반, 이름, 1차시기, 2차시기, 3차시기, Best 기록
  4. 본선 컬럼 순서: 학년, 반, 이름, 1R-1차, 1R-2차, 2R-1차, 2R-2차, Best 기록

구글 앱스크립트(Apps Script) 예선/본선 통합 가이드

구글 스프레드시트 메뉴에서 [확장 프로그램] > [Apps Script]를 누른 뒤, 아래 코드를 붙여넣고 웹 앱으로 배포하세요.

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var prelimSheet = ss.getSheetByName("예선") || ss.getSheets()[0];
  var finalSheet = ss.getSheetByName("본선") || ss.getSheets()[1];
  
  function getPrelimData(sheet) {
    if (!sheet) return [];
    var data = sheet.getDataRange().getValues();
    var result = [];
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (!row[2]) continue; // 이름 없으면 스킵
      result.push({
        grade: String(row[0] || ''),
        cls: String(row[1] || ''),
        name: String(row[2] || ''),
        times: [row[3], row[4], row[5]] // 1,2,3차시기 (Best는 웹에서 자동 계산 또는 row[6] 사용 가능)
      });
    }
    return result;
  }
  
  function getFinalData(sheet) {
    if (!sheet) return [];
    var data = sheet.getDataRange().getValues();
    var result = [];
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      if (!row[2]) continue;
      result.push({
        grade: String(row[0] || ''),
        cls: String(row[1] || ''),
        name: String(row[2] || ''),
        times: [row[3], row[4], row[5], row[6]] // 1R-1, 1R-2, 2R-1, 2R-2
      });
    }
    return result;
  }
  
  var output = {
    preliminary: getPrelimData(prelimSheet),
    final: getFinalData(finalSheet)
  };
  
  return ContentService
    .createTextOutput(JSON.stringify(output))
    .setMimeType(ContentService.MimeType.JSON);
}
